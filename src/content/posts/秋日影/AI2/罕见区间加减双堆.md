// BLOG2 特别罕见的 
// 1. 前缀元素增加 
// 2. 单点删除/增加 
// 3. 查询最大值
// 双堆 + tag 就够了！！！
#include<bits/stdc++.h>
using namespace std;
using ll = long long;

// 群体减去 x 单独删除 v
// 获取最 大 值 单独增加 v
// 依然使用 双堆！！！写法

struct Set {
    priority_queue<ll> A, B;
    ll tag;
    
    void push(ll x) {
        A.push(x-tag);    
    }
    ll top() {
        return A.top() + tag;
    }
    void add(ll x) {
        tag+=x;
    }
    void erase(ll x) {
        B.push(x-tag);
        while(B.size() && B.top() == A.top()) {
            B.pop();
            A.pop();
        }
    }
};

void solve() {
    int n, m, s, t;
    cin >> n >> m >> s >> t;
    vector<vector<int>> a(n+2,vector<int>(m+2));
    for (int i = 1; i <= n; i++)
            for (int j = 1; j <= m; j++)
                    cin >> a[i][j];
    vector<ll> f(n + 2);
    f[s] = a[s][1];
    for (int i =s; ;) {
        int ni = i == n ? 1 : i + 1;
        if (ni == s) break;
        f[ni] = f[i] + a[ni][1];
        i = ni;
    }
    for (int j = 2; j <= m; j++) {
        auto nf = f;
        Set S;
        ll s = 0;
        for (int i=1;i<=n;i++) {
            S.push(f[i]);
            S.add(a[i][j]);
            s += a[i][j];
        }
        for (int i = n; i >= 1; i--) {
            nf[i] = S.top();
            S.add(-a[i][j]);
            S.erase(f[i]);
            S.push(s+f[i]);
        }
        swap(nf, f);
    }
    cout << f[t] << "\n";
}

int main() {
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0);
    solve();
}
