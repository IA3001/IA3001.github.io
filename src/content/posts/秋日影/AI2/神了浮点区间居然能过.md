>实锤假做法：
>
>你爆粗口是可以理解的，这代码纯纯是**面向数据编程的“邪教”**。  
>它的时间复杂度、空间复杂度都没问题，但**浮点数精度在极端深度下必定爆炸**（`1e-12` 的 epsilon 更是槽点满满）。
### Code

```cpp
// BLOG2
// long double 的区间是多少来着 1e4900 精度 1e-18
// 唯一的 坑点 是 tmd 忘了加 l 导致爆 double 了，牛逼！！！
// 忘了加 l 啊啊啊啊啊啊啊啊啊啊啊啊啊啊啊啊
#include<bits/stdc++.h>
using namespace std;
using ll = long long;
using db = long double;

const int N = 5e5 +  5;

int dep[N];
db l[N], r[N];
set<pair<db,int>> L[N];
int tot = 1;

void ins(int u,char c) {
    int v = ++tot;
    dep[v] = dep[u] + 1;
    db mid = (l[u] + r[u]) / 2;
    if (c == 'L') {
        l[v] = l[u];
        r[v] = mid;
    } else {
        l[v] = mid;
        r[v] = r[u];
    }
    L[dep[v]].insert(pair{l[v], v});
//     cout << "ins:" << dep[v] << " " << l[v] << " " << v << "!\n";
}

int query(int u) {
    int d = dep[u];
    auto it = L[d].lower_bound(pair{l[u]+1e-12,99999999});
    if (it == L[d].end()) return -1;
    else return it->second;
}

void solve() {
    int n;
    cin >> n;
    
    dep[1] = 0;
    l[1] = 0;
    r[1] = 1e4900l;
    L[0].insert(pair{0.0,1});
    
    while(n--){
        int op;
        cin >> op;
        if (op == 1) {
            int p;
            char c;
            cin >> p >> c;
            ins(p, c);
        } else {
            int x;
            cin >> x;
            cout << query(x) << '\n';
        }
    }
}

int main() {
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0);
    int T = 1;
//     cin >> T;
    while (T--) solve();
}
```
