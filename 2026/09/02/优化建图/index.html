#include<bits/stdc++.h>
using namespace std;
using ll = long long;
const ll INF = 2e18;

namespace SegG {
    const int N = 4e5 + 5;
    const int NN = N * (__lg(N)+2) + 100;
    
    int n, tot;
    
    int lc[NN], rc[NN];
    int In, Out;

    vector<pair<int,int>> e[NN];

    void add_edge(int u,int v,int w) {
        e[u].push_back({v, w});
    }

    int build_in(int l,int r);
    int build_out(int l,int r);

    // 为了方便 把 init 放在最后吧？
    // 不行， 还是手动吧
    // 不行 还是放前面吧
    void init(int nn) {
        fill(lc,lc+nn,0);
        fill(rc,rc+nn,0);
        for (int i = 1; i <= nn; i++) e[i].clear();

        n = tot = nn;

        In = build_in(1, n);
        Out = build_out(1, n);
    }

    int build_in(int l,int r) {
        int rt;
        if (l == r) {
            rt = l;
        } else {
            rt = ++tot;
            // cout << "in" << tot << ":" << l << " " << r << "\n";
            int mid = (l + r) / 2;
            lc[rt] = build_in(l, mid);
            rc[rt] = build_in(mid+1, r);
            add_edge(rt, lc[rt], 0);
            add_edge(rt, rc[rt], 0);
        }
        return rt;
    }
    
    int build_out(int l,int r) {
        int rt;
        if (l == r) {
            rt = l;
        } else {
            rt = ++tot;
            // cout << "out" << tot << ":" << l << " " << r << "\n";
            int mid = (l + r) / 2;
            lc[rt] = build_out(l, mid);
            rc[rt] = build_out(mid+1, r);
            add_edge(lc[rt], rt, 0);
            add_edge(rc[rt], rt, 0);
        }
        return rt;
    }

    void x_to_range(int i,int l,int r,int jl,int jr,int x,int w) {
        if(jl<=l&&r<=jr) {
            add_edge(x, i, w);
        } else {
            int mid = (l + r) / 2;
            if (jl <= mid) x_to_range(lc[i], l, mid, jl,jr,x,w);
            if (jr > mid) x_to_range(rc[i],mid+1,r,jl,jr,x,w);
        }
    }
    void range_to_x(int i,int l,int r,int jl,int jr,int x,int w) {
        if(jl<=l&&r<=jr) {
            add_edge(i, x, w);
        } else {
            int mid = (l + r) / 2;
            if (jl <= mid) range_to_x(lc[i], l, mid, jl,jr,x,w);
            if (jr > mid) range_to_x(rc[i],mid+1,r,jl,jr,x,w);
        }
    }
    void x_to_r(int x,int l,int r,int w) {
        if (l > r) return;
        x_to_range(In,1,n,l,r,x,w);
    }
    // 更直观！
    void r_to_x(int l,int r,int x,int w) {
        if (l > r) return;
        range_to_x(Out,1,n,l,r,x,w);
    }
    void r_to_r(int l1,int r1,int l2,int r2,int w) {
        int x = ++tot, y = ++tot;
        r_to_x(l1,r1,x,0);
        add_edge(x, y, w);
        x_to_r(y,l2,r2,0);
    }
}

struct CF_786B {
    // 模板题
    // n 个点 q 次加边后 求 s 单源最短路
    // N 1e5 Q 1e5 
    int n, q, s;
    CF_786B() {
        cin >> n >> q >> s;
        SegG::init(n);
        while (q--) {
            int op, u, v, w, l , r;
            cin >> op;
            if (op == 1) {
                cin >> u >> v >> w;
                SegG::add_edge(u, v, w);
            } else if (op == 2) {
                cin >> u >> l >> r >> w;
                SegG::x_to_r(u, l, r, w);
            } else {
                cin >> u >> l >> r >> w;
                SegG::r_to_x(l, r, u, w);
            }
        }
        vector<ll> dist(SegG::tot+1, INF);
        vector<bool> vis(SegG::tot+1);
        priority_queue<pair<ll,int>> Q;
        Q.push({-0, s});
        dist[s] = 0;
        while (Q.size()) {
            int u = Q.top().second; Q.pop();
            if (vis[u]) continue;
            vis[u] = true;
            for (auto [v, w]: SegG::e[u]) {
                if (dist[v] > dist[u] + w) {
                    dist[v] = dist[u] + w;
                    Q.push({-dist[v], v});
                }
            }
        }
        for(int i = 1; i <= n; i++) {
            if (dist[i] == INF) dist[i] = -1;
            cout << dist[i] << " ";
        }cout << "\n";
    }
};

struct LG_P6348 {
    // 模板题 
    // n 个点 m 次 r2r 加边后 求 p 单元最短路(边权1)
    // N 5e5 M 1e5
    int n, m, p;
    LG_P6348() {
        cin >> n >> m >> p;
        SegG::init(n);
        while(m--) {
            int l1, r1, l2, r2;
            cin >> l1 >> r1 >> l2 >> r2;
            SegG::r_to_r(l1, r1, l2, r2, 1);
            SegG::r_to_r(l2, r2, l1, r1, 1);
        }
        deque<int> Q;
        vector<int> dist(SegG::tot+1, 1e9);
        vector<bool> vis(SegG::tot+1);
        Q.push_back(p);
        dist[p] = 0;
        while (Q.size()) {
            int u = Q.front(); Q.pop_front();
            if (vis[u]) continue;
            vis[u] = true;
            for (auto [v, w]: SegG::e[u]) {
                if (dist[v] > dist[u] + w) {
                    dist[v] = dist[u] + w;
                    if (w) Q.push_back(v);
                    else Q.push_front(v);
                }
            }
        }
        for (int i = 1; i <= n; i++) cout << dist[i] << "\n";
    }
};

struct LG_P3588 {
    // 数字大小约束 构造
    // 注意 mle 有点极限
    inline static const int N = 1e5 + 5, M = 2e5 + 5, K = 3e5 + 5, V = 1e9;
    int n, s, m;
    int mn[N * 20], val[N * 20];
    LG_P3588() {
        bool ok = true;
        cin >> n >> s >> m;
        for (int i = 1; i <= s; i++) {
            int p, d;
            cin >> p >> d;
            val[p] = d;
        }
        SegG::init(n);
        for (int i = 1; i <= m; i++) {
            int l, r, k;
            cin >> l >> r >> k;
            vector<int> p(k+2);
            p[0] = l-1;
            for(int i = 1; i <= k;i++) {
                cin >> p[i];
            }
            p[k+1] = r+1;
            int maxv = ++SegG::tot;
            for (int i = 0; i<=k;i++) {
                int l = p[i]+1, r = p[i+1]-1;
                if (l <= r) {
                    SegG::r_to_x(l, r, maxv, 0);
                }
            }
            for(int i = 1; i <= k; i++) {
                SegG::add_edge(maxv, p[i], 1);
            }
        }
        int tot = SegG::tot;
        vector<int> deg(tot + 1);
        for (int u = 1; u <= tot; u++) {
            for (auto [v, w]: SegG::e[u]) {
                deg[v]++;
            }
        }
        int cnt = 0;
        queue<int> Q;
        for (int i = 1; i <= tot; i++) {
            if (!deg[i]) {
                mn[i] = 1;
                Q.push(i);
            }
        }
        while(Q.size()) {
            int u = Q.front(); Q.pop();
            cnt++;
            if (val[u]) {
                ok &= val[u] >= mn[u];
                mn[u] = val[u];
            } else {
                val[u] = mn[u];
            }
            ok &= val[u] <= V;
            for (auto [v, w]: SegG::e[u]) {
                mn[v] = max(mn[v], mn[u] + w);
                if (!--deg[v]) {
                    Q.push(v);
                }
            }
        }
        ok &= cnt == tot;
        if (!ok) {
            cout << "NIE\n";
        } else {
            cout << "TAK\n";
            for (int i = 1; i <= n; i++) {
                cout << val[i] << " ";
            }cout << "\n";
        }
        fill(val, val + 1 + tot, 0);
        fill(mn, mn + 1 + tot, 0);
    }
};

struct CF_1904F {
    inline static const int N = 2e5 + 5, M = 2e5 + 5;
    int n, m, u, v, t, a, b, c;
    int dfn[N], dfncnt, son[N], dep[N], top[N], fa[N], sz[N];
    int rnk[N];
    vector<int> e[N];
    void dfs(int u,int f) {
        sz[u] = 1;
        fa[u] = f;
        dep[u] = dep[f] + 1;
        son[u] = 0;
        for (auto v: e[u]) if(v!=f){
            dfs(v, u);
            sz[u] += sz[v];
            if (sz[v] > sz[son[u]]) son[u] = v;
        }
    }
    void dfs2(int u,int tp) {
        rnk[dfn[u] = ++dfncnt] = u;
        top[u] = tp;
        if (son[u]) dfs2(son[u], tp);
        for (auto v:e[u]) if (v!=fa[u] && v!=son[u]) dfs2(v, v);
    }
    // 【坑点 / 爽点】 
    // 没想到树剖 可以直接批量处理路径
    // 之前一直以为要先求得 lca 才可以
    // 直接就可以维护了， 爽！！！
    void operation(int op,int x,int y,int z) {
        while(top[x] != top[y]) {
            if (dep[top[x]] < dep[top[y]]) swap(x, y);
            if (dfn[z] < dfn[top[x]] || dfn[z] > dfn[x]) {
                if (op == 1) SegG::x_to_r(dfn[z], dfn[top[x]], dfn[x], 1);
                else SegG::r_to_x(dfn[top[x]], dfn[x], dfn[z], 1);
            } else {
                if (op == 1) {
                    SegG::x_to_r(dfn[z], dfn[top[x]], dfn[z]-1, 1);
                    SegG::x_to_r(dfn[z], dfn[z]+1, dfn[x], 1);
                } else {
                    SegG::r_to_x(dfn[top[x]], dfn[z]-1, dfn[z], 1);
                    SegG::r_to_x(dfn[z]+1, dfn[x], dfn[z], 1);
                }
            }
            x = fa[top[x]];
        }
        if (dep[x] < dep[y]) swap(x, y);
        if (dfn[z] < dfn[y] || dfn[z] > dfn[x]) {
            if (op == 1) SegG::x_to_r(dfn[z], dfn[y], dfn[x], 1);
            else SegG::r_to_x(dfn[y], dfn[x], dfn[z], 1);
        } else {
            if (op == 1) {
                SegG::x_to_r(dfn[z], dfn[y], dfn[z]-1, 1);
                SegG::x_to_r(dfn[z], dfn[z]+1, dfn[x], 1);
            } else {
                SegG::r_to_x(dfn[y], dfn[z]-1, dfn[z], 1);
                SegG::r_to_x(dfn[z]+1, dfn[x], dfn[z], 1);
            }
        }
    }
    CF_1904F() {
        dfncnt = 0;
        fill(dfn,dfn+1+n,0);
        for (int i = 1; i <= n; i++) e[i].clear();
        cin >> n >> m;
        SegG::init(n);
        for (int i = 1; i < n; i++) {
            cin >> u >> v;
            e[u].push_back(v);
            e[v].push_back(u);
        }
        dfs(1, 0);
        dfs2(1, 1);
        for (int i = 1; i <= m; i++) {
            cin >> t >> a >> b >> c;
            operation(t, a, b, c);
        }
        int tot = SegG::tot;
        vector<int> deg(tot + 1), val(n + 1);
        for (int u = 1; u <= tot; u++) {
            for (auto [v, w]: SegG::e[u]) {
                deg[v]++;
            }
        }
        queue<int> Q;
        for (int i = 1; i <= tot; i++) {
            if (!deg[i]) Q.push(i); 
        }
        bool ok = true;
        int cnt = 0, num = 0;
        while (Q.size()) {
            int u = Q.front(); Q.pop();
            cnt++;
            if (u <= n) val[rnk[u]] = ++num;
            for (auto [v, w]: SegG::e[u]) {
                if (!--deg[v]) Q.push(v);
            }
        }
        ok &= cnt == tot;
        if (!ok) cout << -1 << "\n";
        else {
            for (int i = 1; i <= n; i++) cout << val[i] << " "; cout << "\n";
        }
    }
};
